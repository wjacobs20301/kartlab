cmake_minimum_required(VERSION 3.20)
project(totk_kart_script_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LOTUSKIT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/totk-runtime/lotuskit" CACHE PATH "Pinned lotuskit source")
set(AS_DIR "${LOTUSKIT_DIR}/subrepo/angelscript")
file(GLOB AS_SOURCES CONFIGURE_DEPENDS "${AS_DIR}/source/*.cpp")
add_library(angelscript_host STATIC ${AS_SOURCES})
target_include_directories(angelscript_host PUBLIC "${AS_DIR}/include")
target_compile_definitions(angelscript_host PUBLIC AS_MAX_PORTABILITY AS_USE_NAMESPACE)
target_compile_options(angelscript_host PRIVATE -Wno-deprecated-declarations)
add_executable(script_harness tests/script_harness.cpp)
target_link_libraries(script_harness PRIVATE angelscript_host)
enable_testing()
add_test(NAME script_contract_and_logic COMMAND script_harness "${CMAKE_CURRENT_SOURCE_DIR}/scripts")
